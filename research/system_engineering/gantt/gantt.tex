% --- Palette (sostituire con i colori aziendali) ----------------------------

% --- Stile globale del diagramma -------------------------------------------
\ganttset{%
  y unit title=0.55cm,
  y unit chart=0.54cm,
  x unit=0.62cm,
  title label font=\scriptsize,
  bar label font=\small,
  group label font=\small\bfseries,
  milestone label font=\small\itshape,
  bar/.append style        ={fill=cBar,  draw=cBar!60!black, rounded corners=1pt},
  group/.append style      ={fill=cGroup,draw=cGroup},
  milestone/.append style  ={fill=cMile, draw=cMile!60!black},
  link/.style              ={->, thick, gray},
  progress label text={\pgfmathprintnumber{#1}\,\%},
}
