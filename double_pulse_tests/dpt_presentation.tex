\section{Motivation}
% ─────────────────────────────────────────────────────────────────────────────

\begin{frame}{Motivation}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Original Design Target (Wind)}
        \begin{itemize}
          \item $I_\text{nom}^\text{inv} = \SI{370}{\ampere}$
          \item $f_\text{pwm}^\text{inv} = \SI{2.5}{\kilo\hertz}$
          \item Overload $\equiv$ wind gust $\Rightarrow$
                $I_\text{max}^\text{inv} \approx I_\text{nom}^\text{inv}$
        \end{itemize}
      \end{block}
      \vspace{4pt}
      \begin{alertblock}{New Requirements (Drive Applications)}
        \begin{itemize}
          \item $I_\text{max}^\text{inv} = 1.4\times I_\text{nom}^\text{inv}
                = \SI{518}{\ampere}$
          \item $f_\text{pwm}^\text{inv} = \SI{4}{\kilo\hertz}$ (hard parallelization)
        \end{itemize}
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Design Constraint}
        \[
          T_\text{j}^\text{design}(t)\Big|_\text{max}
          \;\le\;
          T_\text{j}^\text{max} - \SI{50}{\celsius}
        \]
      \end{block}
      \vspace{4pt}
      \begin{exampleblock}{Conclusion}
        Existing modules (DP650B1700, FF650R17IE4) are no longer adequate.\\[4pt]
        \textbf{Candidate replacement: FF1200R17IP5P} (Infineon,
        1200\,A\,/\,1700\,V, $T_\text{j}^\text{max}=\SI{175}{\celsius}$).
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
